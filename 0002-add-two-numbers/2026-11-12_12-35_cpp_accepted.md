## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode* dummyHead = new ListNode(NULL);
        ListNode* dummyTail= dummyHead;

        int carry = 0;
        while( l1 != NULL && l2 != NULL){
            int sum = carry + l1->val + l2->val;
            int digit = sum %10;
            carry = sum /10;
            
            ListNode* newNode = new ListNode(digit);
            if(dummyHead == NULL){
                dummyHead = newNode;
                dummyTail = newNode;
            }else{
                dummyTail->next = newNode;
                dummyTail = newNode;
            }

            l1 =l1->next;
            l2 = l2->next;

        }
        while(l1 != NULL){
            int sum = carry + l1->val;
            int digit = sum %10;
            carry = sum /10;
            ListNode* newNode = new ListNode(digit);
            dummyTail->next = newNode;
            dummyTail = newNode;
            l1 = l1->next;
        }
        while(l2 != NULL){
            int sum = carry + l2->val;
            int digit = sum %10;
            carry = sum /10;
            ListNode* newNode = new ListNode(digit);
            dummyTail->next = newNode;
            dummyTail = newNode;
            l2 = l2->next;
        }
        while(carry != 0){
            int sum = carry;
            int digit = sum %10;
            carry = sum /10;
            ListNode* newNode = new ListNode(digit);
            dummyTail->next = newNode;
            dummyTail = newNode;

        }
        return dummyHead->next;
    }
};
```
