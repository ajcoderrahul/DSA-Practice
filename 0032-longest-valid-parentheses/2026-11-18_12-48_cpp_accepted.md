## Solution

```cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        
        if(s.size() == 0){
            return 0;
        }
        int ans = 0,length = 0;
        stack<int>st;
        st.push(-1);

        for(int i = 0;i < s.size();i++){
            if(s[i]== '('){
                st.push(i);
            }else{
                st.pop();
                if(!st.empty()){
                    
                    length = i- st.top();
                    ans = max(length,ans);

                }else{
                    st.push(i);
                }
                 
            }
            
        }
        return ans;
    }
};
```
