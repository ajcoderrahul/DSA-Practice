## Solution

```cpp
class Solution {
public:
    int trap(vector<int>& height) {
        int l = 0,r = height.size()-1;
        int lmax = 0,rmax = 0,water = 0;
        while(l<r){
            if(height[l] < height[r] ){
                lmax = max(height[l],lmax);
                water += lmax-height[l];
                l++;
            }else{
                rmax = max(height[r],rmax);
                water += rmax- height[r];
                r--; 
            }
            
        }
        return water;
    }
};
```
