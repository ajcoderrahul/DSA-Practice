## Solution

```cpp
class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int>a(n,-1);
        stack<int>st;
        
        
        for(int i =2*n-1;i>=0;i--){
            while(!st.empty() && st.top() <= nums[i%n]){
                st.pop();
            }
            if(i < n){
                if(!st.empty()){
                a[i] = st.top();
               }
            }
               
             st.push(nums[i%n]); 
            
        }
        
        return a;
    }
};
```
