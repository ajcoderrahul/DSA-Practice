## Solution

```cpp
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        stack<int>st;
        st.push(0);
       vector<int> a(temperatures.size(), 0);
        for(int i = 1;i<temperatures.size();i++){
            while(!st.empty() && temperatures[i] > temperatures[st.top()]){
                int index = st.top();
                st.pop();
                
                a[index] = i - index;
            }
            st.push(i);

        }
        return a;
    }
};
```
