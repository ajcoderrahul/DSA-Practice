## Solution

```cpp
class Solution {
public:
    int minAddToMakeValid(string s) {
        int count = 0,add = 0;
        for(int  i = 0;i<s.size();i++){
            if(s[i] == '('){
                count++;
            }else
            if(s[i] == ')'){
                add++;
                if(count > 0){
                   count -= 1;
                   add--;
                }
                
            }

        }
        return add + count;
    }
};
```
