## Solution

```cpp
class Solution {
public:
    void sortColors(vector<int>& nums) {
      int n =nums.size();
      int j = 0,i = 0;
      int k = n-1; 
      while(i <= k){
        if(nums[i] == 0){
            swap(nums[j],nums[i]);
            j++;
            i++;
        }else
        if(nums[i]== 2){
            swap(nums[i],nums[k]);
            k--;           
            
        }else{
            i++;
        }
        
      }
    
    }
};
```
