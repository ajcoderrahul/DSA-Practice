## Solution

```cpp
class Solution {
public:
    
    
    int rob(vector<int>& nums) {
        int i = 0,j = 0;
        for(int x : nums){
            int k = max(i,j+x);
            j = i;
            i = k;
        }
        return i;
        
    }
};
```
