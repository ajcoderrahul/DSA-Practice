## Solution

```cpp
class Solution {
public:
    int lowerBond(vector <int>&arr,int x){
        int s = 0;
        int e = arr.size()-1;
        int ans = e;
        while(s<= e){
            int mid = s+(e-s)/2;
            if(arr[mid]>= x){
                ans =mid;
                e= mid -1;
            }else {
                s = mid +1;
            }
        }
        return ans;
    }
    vector<int>binarySearch(vector<int>& arr, int k, int x){
        int h = lowerBond(arr,x);
        int l = h-1;
        while(k--){
            if(l<0){
                h++;
            }else if(arr.size()<=h){
                l--;
            }else if(x-arr[l] <= arr[h]-x){
                l--;
            }else{
                h++;
            }
        }
       return vector<int>(arr.begin()+l+1,arr.begin()+h);

    }
    vector<int> findClosestElements(vector<int>& arr, int k, int x) {
        return binarySearch(arr,k,x);
        
    }
};
```
