## Solution

```cpp
class Solution {
public:
    int peakindex(vector <int> arr){
        int s = 0;
        int e = arr.size()-1;
        while(s <e){
            int mid = s+(e-s)/2;
            if(arr[mid] < arr[mid+1]){
                s = mid+1;
            }else{
               e=mid;
            }
        }
        return e;
    }
    int peakIndexInMountainArray(vector<int>& arr) {
        return peakindex(arr);
    }
};
```
